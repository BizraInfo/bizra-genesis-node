rustimpl IhsanGate {
    fn check_invariants(json: &Value, invariants: &[Invariant]) -> f32 {
        if invariants.is_empty() { return 1.0; }
        
        let mut passed = 0;
        let total = invariants.len();
        
        for inv in invariants {
            if Self::evaluate_invariant(json, inv) {
                passed += 1;
            }
        }
        
        passed as f32 / total as f32
    }
    
    fn evaluate_invariant(json: &Value, inv: &Invariant) -> bool {
        match inv {
            Invariant::Always => true,
            Invariant::Range { path, min, max } => {
                Self::get_value_at_path(json, path)
                    .and_then(|v| v.as_f64())
                    .map(|v| v >= *min && v <= *max)
                    .unwrap_or(false)
            },
            Invariant::Regex { path, pattern } => {
                use regex::Regex;
                let re = Regex::new(pattern).ok()?;
                Self::get_value_at_path(json, path)
                    .and_then(|v| v.as_str())
                    .map(|s| re.is_match(s))
                    .unwrap_or(false)
            },
            Invariant::Monotone { path } => {
                // Check if array at path is monotonically increasing
                Self::get_value_at_path(json, path)
                    .and_then(|v| v.as_array())
                    .map(|arr| {
                        arr.windows(2).all(|w| {
                            let a = w[0].as_f64().unwrap_or(f64::NEG_INFINITY);
                            let b = w[1].as_f64().unwrap_or(f64::NEG_INFINITY);
                            a <= b
                        })
                    })
                    .unwrap_or(false)
            }
        }
    }
    
    fn get_value_at_path<'a>(json: &'a Value, path: &str) -> Option<&'a Value> {
        let parts: Vec<&str> = path.split('.').collect();
        let mut current = json;
        
        for part in parts {
            current = current.get(part)?;
        }
        
        Some(current)
    }
}

impl Contract {
    pub fn validate(&self) -> Result<(), ValidationError> {
        if self.token_budget == 0 {
            return Err(ValidationError::InvalidTokenBudget);
        }
        // Validate schema_json is valid JSON
        serde_json::from_str::<Value>(&self.schema_json)
            .map_err(|e| ValidationError::InvalidSchema(e.to_string()))?;
        Ok(())
    }
}

impl Candidate {
    pub fn validate_scores(&self) -> Result<(), ValidationError> {
        let s = &self.scores;
        if !(0.0..=1.0).contains(&s.accuracy) ||
           !(0.0..=1.0).contains(&s.safety) ||
           !(0.0..=1.0).contains(&s.efficiency) ||
           !(0.0..=1.0).contains(&s.ihsan) {
            return Err(ValidationError::ScoreOutOfRange);
        }
        Ok(())
    }
}

#[derive(thiserror::Error, Debug)]
pub enum ValidationError {
    #[error("Invalid token budget: must be > 0")]
    InvalidTokenBudget,
    #[error("Invalid schema JSON: {0}")]
    InvalidSchema(String),
    #[error("Score out of valid range [0.0, 1.0]")]
    ScoreOutOfRange,
}

powershell# scripts/summarize_criterion.ps1
param(
    [string]$Root = "target/criterion",
    [switch]$Verbose
)

$ErrorActionPreference = "Stop"

Write-Host "🔍 Scanning criterion results in: $Root" -ForegroundColor Cyan

$files = Get-ChildItem -Recurse -Path $Root -Filter "estimates.json" -ErrorAction SilentlyContinue

if ($files.Count -eq 0) {
    Write-Host "❌ No benchmark results found. Run 'cargo bench' first." -ForegroundColor Red
    exit 1
}

Write-Host "📊 Found $($files.Count) benchmark results" -ForegroundColor Green
$rows = @()

foreach ($f in $files) {
    try {
        $json = Get-Content $f.FullName -Raw | ConvertFrom-Json
        
        # Extract metrics
        $mean_ns = [double]$json.mean.point_estimate
        $std_ns = [double]$json.std_dev.point_estimate
        
        # Parse benchmark name from path
        $parts = $f.Directory.FullName.Split([IO.Path]::DirectorySeparatorChar)
        $fn = $parts[$parts.Length - 2]
        $grp = $parts[$parts.Length - 3]
        
        # Skip if it's a comparison directory
        if ($fn -eq "new" -or $fn -eq "base" -or $fn -eq "change") {
            continue
        }
        
        $name = "$grp/$fn"
        
        $rows += [PSCustomObject]@{
            Benchmark = $name
            Mean_ns = [math]::Round($mean_ns, 2)
            StdDev_ns = [math]::Round($std_ns, 2)
            Mean_us = [math]::Round($mean_ns / 1000.0, 2)
            Mean_ms = [math]::Round($mean_ns / 1e6, 3)
            Throughput = [math]::Round(1e9 / $mean_ns, 0)
        }
        
        if ($Verbose) {
            Write-Host "  ✓ Processed: $name" -ForegroundColor Gray
        }
    } catch {
        Write-Host "  ⚠️  Failed to process: $($f.FullName)" -ForegroundColor Yellow
    }
}

# Sort and display
$rows = $rows | Sort-Object Benchmark

Write-Host "`n📈 Benchmark Results:" -ForegroundColor Cyan
Write-Host ("=" * 120) -ForegroundColor DarkGray

$rows | Format-Table -AutoSize @{
    Label = "Benchmark"; Expression = { $_.Benchmark }; Width = 40
}, @{
    Label = "Mean (ns)"; Expression = { $_.Mean_ns }; Width = 15; Alignment = "Right"
}, @{
    Label = "StdDev (ns)"; Expression = { $_.StdDev_ns }; Width = 15; Alignment = "Right"
}, @{
    Label = "Mean (μs)"; Expression = { $_.Mean_us }; Width = 12; Alignment = "Right"
}, @{
    Label = "Mean (ms)"; Expression = { $_.Mean_ms }; Width = 12; Alignment = "Right"
}, @{
    Label = "Ops/sec"; Expression = { "{0:N0}" -f $_.Throughput }; Width = 15; Alignment = "Right"
}

# Save CSV
$csvPath = Join-Path $Root "summary.csv"
$rows | Export-Csv -Path $csvPath -NoTypeInformation -Encoding UTF8
Write-Host "`n💾 Summary saved to: $csvPath" -ForegroundColor Green

# Performance analysis
Write-Host "`n🎯 Performance Insights:" -ForegroundColor Cyan
$slowest = $rows | Sort-Object Mean_ms -Descending | Select-Object -First 3
$fastest = $rows | Sort-Object Mean_ms | Select-Object -First 3

Write-Host "   Slowest operations:" -ForegroundColor Yellow
$slowest | ForEach-Object { Write-Host "     • $($_.Benchmark): $($_.Mean_ms) ms" -ForegroundColor Gray }

Write-Host "   Fastest operations:" -ForegroundColor Green
$fastest | ForEach-Object { Write-Host "     • $($_.Benchmark): $($_.Mean_us) μs" -ForegroundColor Gray }

# Highlight potential optimization targets
$targets = $rows | Where-Object { $_.Mean_ms -gt 1.0 }
if ($targets) {
    Write-Host "`n⚠️  Optimization Targets (>1ms):" -ForegroundColor Yellow
    $targets | ForEach-Object { Write-Host "     • $($_.Benchmark)" -ForegroundColor Gray }
}

bash#!/usr/bin/env bash
# scripts/summarize_criterion.sh

set -euo pipefail

ROOT="${1:-target/criterion}"

echo "🔍 Scanning criterion results in: $ROOT"

if ! command -v jq &> /dev/null; then
    echo "❌ jq is required. Install with: sudo apt-get install jq"
    exit 1
fi

tmp=$(mktemp)
count=0

find "$ROOT" -name "estimates.json" | while read -r f; do
    # Skip comparison directories
    if [[ "$f" =~ (new|base|change)/estimates.json$ ]]; then
        continue
    fi
    
    mean_ns=$(jq -r '.mean.point_estimate' "$f")
    std_ns=$(jq -r '.std_dev.point_estimate' "$f")
    
    # Extract benchmark name
    d=$(dirname "$f")
    fn=$(basename "$d")
    grp=$(basename "$(dirname "$d")")
    
    mean_us=$(echo "scale=2; $mean_ns / 1000" | bc -l)
    mean_ms=$(echo "scale=3; $mean_ns / 1000000" | bc -l)
    ops_sec=$(echo "scale=0; 1000000000 / $mean_ns" | bc -l)
    
    printf "%s,%s,%.2f,%.2f,%.2f,%.3f,%s\n" \
        "$grp" "$fn" "$mean_ns" "$std_ns" "$mean_us" "$mean_ms" "$ops_sec"
    ((count++))
done > "$tmp"

echo "📊 Found $count benchmark results"
echo ""
echo "📈 Benchmark Results:"
echo "=================================================================================================="

printf "%-40s %15s %15s %12s %12s %15s\n" \
    "Benchmark" "Mean (ns)" "StdDev (ns)" "Mean (μs)" "Mean (ms)" "Ops/sec"
printf "%-40s %15s %15s %12s %12s %15s\n" \
    "----------------------------------------" "---------------" "---------------" "------------" "------------" "---------------"

awk -F, '{printf "%-40s %15.2f %15.2f %12.2f %12.3f %15s\n", $1"/"$2, $3, $4, $5, $6, $7}' "$tmp"

# Save CSV
out="$ROOT/summary.csv"
mkdir -p "$(dirname "$out")"
echo "Benchmark,Function,Mean_ns,StdDev_ns,Mean_us,Mean_ms,Ops_sec" > "$out"
cat "$tmp" >> "$out"
rm "$tmp"

echo ""
echo "💾 Summary saved to: $out"

rust// synthesis_orchestrator/benches/e2e_pprof.rs
use criterion::{criterion_group, criterion_main, Criterion, BatchSize};
use serde_json::json;
use synthesis_orchestrator::{
    adapters::mock::MockAdapter,
    orchestrator::SynthesisOrchestrator,
    types::*,
};

#[cfg(not(target_env = "msvc"))]
use pprof::ProfilerGuard;

fn bench_pprof(c: &mut Criterion) {
    #[cfg(not(target_env = "msvc"))]
    {
        let mut group = c.benchmark_group("e2e-pprof");
        group.sample_size(100);
        
        group.bench_function("profile_500_iterations", |b| {
            b.iter_batched(
                || {
                    let adapter = MockAdapter::default();
                    let orch = SynthesisOrchestrator::new(adapter, 0.85).with_fanout(3);
                    let task = Task {
                        task_id: "pprof".into(),
                        prompt: "profile".into(),
                        context: json!({"lang": "en"}),
                        examples: None,
                    };
                    let mut contract = Contract::example();
                    contract.schema_json = r#"{"type":"object","required":["name","age"]}"#.into();
                    let routes = vec![
                        Route::example1(),
                        Route::example2(),
                        Route::example3(),
                    ];
                    (orch, task, contract, routes)
                },
                |(mut orch, task, contract, routes)| {
                    // Profile 500 iterations
                    let guard = ProfilerGuard::new(100).expect("pprof guard");
                    
                    let rt = tokio::runtime::Runtime::new().unwrap();
                    for _ in 0..500 {
                        rt.block_on(async {
                            let _ = orch.synthesize(&task, &contract, &routes).await.unwrap();
                        });
                    }
                    
                    if let Ok(report) = guard.report().build() {
                        let file = std::fs::File::create("flamegraph.svg").unwrap();
                        let _ = report.flamegraph(file);
                    }
                },
                BatchSize::SmallInput,
            );
        });
        
        group.finish();
        
        println!("\n🔥 Flamegraph generated: flamegraph.svg");
        println!("   View with: firefox flamegraph.svg");
    }
    
    #[cfg(target_env = "msvc")]
    {
        println!("⚠️  pprof profiling not available on MSVC. Use cargo-flamegraph instead:");
        println!("   cargo install flamegraph");
        println!("   cargo flamegraph --bench e2e_pipeline");
    }
}
